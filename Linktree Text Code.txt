<script>
const canvas = document.getElementById('matrixCanvas');
const ctx = canvas.getContext('2d');

// Fit canvas to full window size
function resizeCanvas() {
canvas.width = window.innerWidth;
canvas.height = window.innerHeight;
}

window.addEventListener('resize', resizeCanvas);
resizeCanvas();

// --- EDITABLE MATRIX TREE CONFIGURATION ---
const matrixChars = "0101011001101011001100101010101010010101101"; // Symbols used
const fontSize = 14; 
const streamSpeed = 25; // Speed multiplier (higher = faster stream update)
        
// Setup initial structural coordinate arrays for structural tree look
const columns = Math.floor(canvas.width / fontSize);
const yPositions = Array(columns).fill(0).map(() => Math.random() * -canvas.height);

function drawMatrixTree() {
// Slight opacity black fade creates trailing effect
ctx.fillStyle = 'rgba(0, 0, 0, 0.08)';
ctx.fillRect(0, 0, canvas.width, canvas.height);

for (let i = 0; i < yPositions.length; i++) {
const char = matrixChars[Math.floor(Math.random() * matrixChars.length)];
const x = i * fontSize;
const y = yPositions[i];

// Calculate the "Tree Silhouette Zone" in center screen
const centerX = canvas.width / 2;
const distFromCenter = Math.abs(x - centerX);
                
// Form trunk narrowing towards top, widening at base
const trunkWidth = 40 + (y * 0.15); 
// Form branches zone radiating outwards at mid-height
const branchWidth = Math.sin((y / canvas.height) * Math.PI) * (canvas.width * 0.35);
const acceptableTreeZone = Math.max(trunkWidth, branchWidth);

// EDIT THE COLOR OF THE MATRIX STRANDS HERE
if (distFromCenter < acceptableTreeZone) {
ctx.fillStyle = '#00ff99'; // Bright inner matrix green
} 

else {
ctx.fillStyle = 'rgba(0, 50, 20, 0.15)'; // Dim, heavily faded outer code
}

ctx.font = fontSize + 'px monospace';
ctx.fillText(char, x, y);

// Reset stream to top randomly once it hits the bottom
if (y > canvas.height && Math.random() > 0.985) {
yPositions[i] = 0;
}

else {
yPositions[i] += fontSize; // Advance vertical cascade stream down
}

// Run animation frame loop
setInterval(drawMatrixTree, streamSpeed);

</script>
